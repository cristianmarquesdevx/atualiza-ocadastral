function salvarCache() { const cnes = document.getElementById('cnes').value; const selectDisplay = document.getElementById('selectDisplay'); const cnesLabel = selectDisplay.textContent !== 'Selecione a Unidade' ? selectDisplay.textContent : ''; const dataEmissao = document.getElementById('dataEmissao').value; const linhas = document.querySelectorAll('#tbodyProfissionais tr'); const profissionais = []; linhas.forEach(tr => { const acao = tr.querySelector('.prof-acao')?.value || 'Inclusão'; const nome = tr.querySelector('.prof-nome')?.value || ''; const cpf = tr.querySelector('.prof-cpf')?.value || ''; const cbo = tr.querySelector('.prof-cbo')?.value || ''; const conselho = tr.querySelector('.prof-conselho')?.value || ''; const registro = tr.querySelector('.prof-registro')?.value || ''; const uf = tr.querySelector('.prof-ufconselho')?.value || ''; const cargo = tr.querySelector('.prof-cargo')?.value || ''; const vinculo = tr.querySelector('.prof-vinculo')?.value || ''; const carga = tr.querySelector('.prof-carga')?.value || ''; const setor = tr.querySelector('.prof-setor')?.value || ''; profissionais.push({ acao, nome, cpf, cbo, conselho, registro, uf, cargo, vinculo, carga, setor }); }); const cache = { cnes, cnesLabel, dataEmissao, profissionais }; localStorage.setItem('ficha_cnes_cache', JSON.stringify(cache)); } function restaurarCache() { const cacheStr = localStorage.getItem('ficha_cnes_cache'); if (!cacheStr) return false; try { const cache = JSON.parse(cacheStr); if (cache.cnes) { document.getElementById('cnes').value = cache.cnes; document.getElementById('cnes').classList.add('unidade-selecionada'); if (cache.cnesLabel) { document.getElementById('selectDisplay').textContent = cache.cnesLabel; document.getElementById('selectDisplay').dataset.value = cache.cnes; document.getElementById('tipoUnidade').value = cache.cnes; } } if (cache.dataEmissao) { document.getElementById('dataEmissao').value = cache.dataEmissao; } // Limpar tbody const tbody = document.getElementById('tbodyProfissionais'); tbody.innerHTML = ''; if (cache.profissionais && cache.profissionais.length > 0) { cache.profissionais.forEach(dados => { const tr = criarLinha(dados); tbody.appendChild(tr); }); // Atualizar contador contadorLinhas = cache.profissionais.length + 1; } else { // Se não houver profissionais, adicionar uma linha vazia adicionarLinha(); } return true; } catch (e) { console.error('Erro ao restaurar cache', e); return false; } }
